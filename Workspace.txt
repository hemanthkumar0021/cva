from pyspark.sql.functions import col, current_date, date_sub

# 1. Read the source table
df_srp_full = spark.table("med_cube_nprd.iex_gold.iex_schedule_blue")

# 2. Filter the data for the last 14 days
# REPLACE "your_date_column" with the actual date or timestamp column in your table
df_srp_filtered = df_srp_full.filter(col("your_date_column") >= date_sub(current_date(), 14))

# 3. Write the filtered data
df_srp_filtered.write.mode("overwrite").parquet(iex_schedule_blue)

print(f"Data written to: {iex_schedule_blue}")
print(f"Records written: {df_srp_filtered.count()}")
