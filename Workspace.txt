from pyspark.sql.functions import col, current_date, date_sub

# 1. Fetch metadata for the target tables in a single query
# Either specify the 4 destination tables in an IN clause or rely on app_id/control_main_id/load_type
query = """
SELECT 
    source_table_name,
    destination_table,
    sql_query,
    where_condition
FROM med_cube_nprd.configuration_schema.control_trans_dev
WHERE app_id = 'intradiem_fo_optum'
  AND control_main_id = 3
  AND load_type = 'full_load'
  -- If you only want 4 specific tables out of the app_id, specify them here:
  -- AND destination_table IN ('table1', 'table2', 'table3', 'optum_userlastlogin')
"""

tables_df = spark.sql(query)
table_records = tables_df.collect()

print(f"Total tables to process: {len(table_records)}")

# 2. Iterate dynamically over each table record
for row in table_records:
    src_table = row["source_table_name"]
    dest_table = row["destination_table"]
    columns_string = row["sql_query"]
    row_date_column = row["where_condition"]

    print(f"\n--- Processing: Source [{src_table}] -> Destination [{dest_table}] ---")

    # Define source/target table paths
    current_table_temp = f"med_cube_nprd.intradiem_fo_gold.{dest_table}"

    # Handle optional date filtering if where_condition is provided
    if row_date_column and row_date_column.strip():
        sql_statement = f"""
            SELECT {columns_string}
            FROM {current_table_temp}
            WHERE {row_date_column} >= date_sub(current_date(), 15)
              AND {row_date_column} <= current_date()
        """
    else:
        # Fallback if no date column filter is defined
        sql_statement = f"SELECT {columns_string} FROM {current_table_temp}"

    print(f"Executing: {sql_statement}")
    df_srp = spark.sql(sql_statement)
